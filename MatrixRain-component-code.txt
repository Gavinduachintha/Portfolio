import { useEffect, useRef, useState } from 'react';
import PropTypes from 'prop-types';

const MatrixRain = ({
  intensity = 0.5,
  speed = 1,
  characterSet = 'mixed',
  colors = ['#3B82F6', '#8B5CF6', '#06B6D4'],
  className = '',
  style = {},
}) => {
  const canvasRef = useRef(null);
  const animationRef = useRef(null);
  const dropsRef = useRef([]);
  const [isReduced, setIsReduced] = useState(false);

  useEffect(() => {
    // Check for reduced motion preference
    const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    setIsReduced(mediaQuery.matches);

    const handleChange = (e) => setIsReduced(e.matches);
    mediaQuery.addEventListener('change', handleChange);
    
    return () => mediaQuery.removeEventListener('change', handleChange);
  }, []);

  useEffect(() => {
    if (isReduced) return;

    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    
    // Set canvas size to window size
    const resizeCanvas = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      initializeDrops();
    };

    // Determine character set
    const getCharacterSet = () => {
      const sets = {
        binary: '01',
        hex: '0123456789ABCDEF',
        tech: '01アイウエオカキクケコサシスセソタチツテトナニヌネノハヒフヘホマミムメモヤユヨラリルレロワヲン',
        mixed: '01アイウエオ<>[]{}()#@$%^&*',
      };
      return sets[characterSet] || characterSet;
    };

    const chars = getCharacterSet();

    // Detect mobile device
    const isMobile = window.innerWidth < 768;
    const isLowEnd = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
    
    // Adjust performance based on device
    const columnWidth = isMobile ? 20 : 16;
    const baseIntensity = isMobile ? intensity * 0.5 : intensity;
    const targetFPS = isMobile && isLowEnd ? 24 : 60;
    const frameInterval = 1000 / targetFPS;
    
    let lastFrameTime = 0;

    // Initialize drops
    const initializeDrops = () => {
      const columns = Math.floor(canvas.width / columnWidth);
      const numberOfDrops = Math.floor(columns * baseIntensity);
      
      dropsRef.current = Array.from({ length: numberOfDrops }, () => ({
        x: Math.random() * canvas.width,
        y: Math.random() * -canvas.height,
        speed: (Math.random() * 2 + 1) * speed * (isMobile ? 0.8 : 1),
        length: Math.random() * 15 + 10,
        opacity: Math.random() * 0.5 + 0.5,
        color: colors[Math.floor(Math.random() * colors.length)],
        chars: Array.from({ length: 20 }, () => 
          chars.charAt(Math.floor(Math.random() * chars.length))
        ),
      }));
    };

    // Animation loop
    const animate = (currentTime) => {
      animationRef.current = requestAnimationFrame(animate);

      // Frame rate limiting
      if (currentTime - lastFrameTime < frameInterval) {
        return;
      }
      lastFrameTime = currentTime;

      // Clear canvas with fade effect
      ctx.fillStyle = 'rgba(0, 0, 0, 0.05)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // Draw each drop
      dropsRef.current.forEach((drop) => {
        const { x, y, speed: dropSpeed, length, opacity, color, chars: dropChars } = drop;

        // Draw characters in the drop
        for (let i = 0; i < length; i++) {
          const charY = y + i * columnWidth;
          
          if (charY > 0 && charY < canvas.height) {
            // Calculate opacity based on position in drop and distance from bottom
            const positionInDrop = i / length;
            const distanceFromBottom = 1 - (charY / canvas.height);
            const fadeOut = Math.min(distanceFromBottom * 2, 1);
            const charOpacity = opacity * (1 - positionInDrop * 0.7) * fadeOut;

            // Brightest character at the head
            const isHead = i === 0;
            const alpha = isHead ? Math.min(charOpacity * 1.5, 1) : charOpacity;

            // Set glow effect for head character
            if (isHead && !isMobile) {
              ctx.shadowBlur = 10;
              ctx.shadowColor = color;
            } else {
              ctx.shadowBlur = 0;
            }

            ctx.fillStyle = color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
            ctx.font = `${isMobile ? 14 : 12}px monospace`;
            ctx.fillText(dropChars[i % dropChars.length], x, charY);
          }
        }

        // Update drop position
        drop.y += dropSpeed;

        // Reset drop when it goes off screen
        if (drop.y - length * columnWidth > canvas.height) {
          drop.y = Math.random() * -100;
          drop.x = Math.random() * canvas.width;
          drop.speed = (Math.random() * 2 + 1) * speed * (isMobile ? 0.8 : 1);
          drop.color = colors[Math.floor(Math.random() * colors.length)];
          drop.opacity = Math.random() * 0.5 + 0.5;
          
          // Randomize characters occasionally
          if (Math.random() > 0.7) {
            drop.chars = Array.from({ length: 20 }, () => 
              chars.charAt(Math.floor(Math.random() * chars.length))
            );
          }
        }
      });
    };

    // Initialize
    resizeCanvas();
    window.addEventListener('resize', resizeCanvas);
    animationRef.current = requestAnimationFrame(animate);

    // Cleanup
    return () => {
      window.removeEventListener('resize', resizeCanvas);
      if (animationRef.current) {
        cancelAnimationFrame(animationRef.current);
      }
    };
  }, [intensity, speed, characterSet, colors, isReduced]);

  if (isReduced) {
    return null;
  }

  return (
    <canvas
      ref={canvasRef}
      className={className}
      style={{
        position: 'fixed',
        top: 0,
        left: 0,
        width: '100%',
        height: '100%',
        pointerEvents: 'none',
        zIndex: -1,
        ...style,
      }}
      aria-hidden="true"
    />
  );
};

MatrixRain.propTypes = {
  intensity: PropTypes.number,
  speed: PropTypes.number,
  characterSet: PropTypes.oneOfType([
    PropTypes.oneOf(['binary', 'hex', 'tech', 'mixed']),
    PropTypes.string,
  ]),
  colors: PropTypes.arrayOf(PropTypes.string),
  className: PropTypes.string,
  style: PropTypes.object,
};

export default MatrixRain;
